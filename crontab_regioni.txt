# >>> METEO (blocco gestito da deploy.sh: aggiorna SOLO fra questi due marcatori) >>>
# Crontab dei download e parse dei dati idro-meteo regionali (Raspberry).
# I percorsi degli script derivano da BASE (la cartella del progetto): gli script trovano da soli il
# progetto e scrivono nelle cartelle delle regioni. Le cartelle 'logs' si creano da sole.
# Ogni script scarica, elabora e, a elaborazione riuscita, cancella i file grezzi.
# Nel crontab il carattere '%' va sempre escapato come '\%'.
#
# Cadenze, con i minuti sfalsati per non sovrapporre i run:
#   Bolzano ogni 10 min (fonte a snapshot), Trentino ogni 15, Veneto ogni 30 (+ run 'ieri'
#   giornaliero), Friuli ogni 60.

BASE=/home/manuel/METEO
PYTHON=/usr/bin/python3

# --- ALTO ADIGE / BOLZANO: ogni 10 minuti, 24h
2,12,22,32,42,52 * * * * mkdir -p $BASE/BZ/logs && $PYTHON $BASE/dati_altoadige.py >> $BASE/BZ/logs/download-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1

# --- TRENTINO: ogni 15 minuti, 24h
4,19,34,49 * * * * mkdir -p $BASE/TN/logs && $PYTHON $BASE/dati_trentino.py >> $BASE/TN/logs/download-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1

# --- VENETO (ARPAV): ogni 30 minuti, 24h + run 'ieri' una volta (dati arrivati in ritardo)
# flock -n salta il run se il precedente non ha finito: VEN scarica centinaia di stazioni una a una
# e puo' sforare i 30 minuti; un run ucciso a meta' lascerebbe grezzi orfani.
6,36 * * * * mkdir -p $BASE/VEN/logs && flock -n $BASE/VEN/.lock $PYTHON $BASE/dati_veneto.py -date t >> $BASE/VEN/logs/download-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1
5 6  * * *   mkdir -p $BASE/VEN/logs && flock -n $BASE/VEN/.lock $PYTHON $BASE/dati_veneto.py -date y >> $BASE/VEN/logs/download-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1

# --- FRIULI VENEZIA GIULIA: ogni 60 minuti, 24h
8 * * * * mkdir -p $BASE/FVG/logs && $PYTHON $BASE/dati_fvg.py >> $BASE/FVG/logs/download-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1

# --- LOMBARDIA (ARPA Lombardia): ogni 30 minuti, 24h. Dato a 10 min, finestra 3h: ogni run riscarica
# le ultime 3 ore, quindi anche 5 run falliti di fila non lasciano buchi. Il portale Socrata limita le
# richieste anonime (HTTP 429): lo script ritenta con pause e, se non basta, esce senza scrivere.
# flock -n: mai due istanze insieme (con i ritentativi un run puo' durare qualche minuto).
0,30 * * * * mkdir -p $BASE/LOM/logs && flock -n $BASE/LOM/.lock $PYTHON $BASE/dati_lombardia.py >> $BASE/LOM/logs/download-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1

# --- EMILIA-ROMAGNA (ARPAE / Allerta Meteo): ogni 15 minuti, 24h (snapshot a 15 min)
9,24,39,54 * * * * mkdir -p $BASE/ER/logs && $PYTHON $BASE/dati_emilia_romagna.py >> $BASE/ER/logs/download-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1

# --- QUALITA' DELL'ARIA (dati_aria.py): ogni ora. Un run per tutte le reti (VEN/LOM/BZ/TN/ER); ogni
# fonte scarica una finestra di piu' giorni, quindi il run ricuce da solo i buchi. Scrive in AQ_<REG>.
12 * * * * mkdir -p $BASE/logs && $PYTHON $BASE/dati_aria.py -r all >> $BASE/logs/aria-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1

# --- IDROMETRIA PROTEZIONE CIVILE (dati_pc.py): stazioni idrometriche con coordinate
#   PC_FVG: snapshot (measures/latest), ogni 20 min, la serie si accumula.
#   PC_TN : ogni serie porta 7 giorni di storico, bastano ogni 2 ore (131 serie/run).
7,27,47 * * * * mkdir -p $BASE/logs && $PYTHON $BASE/dati_pc.py -r FVG >> $BASE/logs/pc-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1
17 */2 * * *   mkdir -p $BASE/logs && $PYTHON $BASE/dati_pc.py -r TN >> $BASE/logs/pc-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1

# --- MAREE E METEO-MARINO (dati_mare.py -r VEN,ER,FVG): snapshot ogni 10 min, la cadenza della fonte
#   MAR_VEN (CPSM Venezia) e MAR_ER (ARPAE) aggiornano ogni ~10 min; MAR_FVG (OGS Trieste ERDDAP) e'
#   oraria, ma si interroga ogni 10 min per ridurre il ritardo (richiesta piccola, finestra 2 giorni).
# ACQUE MARINO-COSTIERE (dati_mare.py -r ACQVEN): ARPAV, sorgente oraria con ~1h30 di ritardo, un solo
#   run l'ora, a :50.
# AIPO / BACINO DEL PO (dati_pc.py -r PO): ogni serie porta 7 giorni, quindi ogni 3 ore. Accesso
#   anonimo, SSL incompleto gestito nello script. Pubblicato completo, anche i doppioni con ER/LOM.
3,13,23,33,43,53 * * * * mkdir -p $BASE/logs && $PYTHON $BASE/dati_mare.py -r VEN,ER,FVG >> $BASE/logs/mare-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1
50 * * * *    mkdir -p $BASE/logs && $PYTHON $BASE/dati_mare.py -r ACQVEN >> $BASE/logs/mare-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1
37 */3 * * *   mkdir -p $BASE/logs && $PYTHON $BASE/dati_pc.py -r PO >> $BASE/logs/pc-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1

# --- PIEMONTE (dati_pc.py -r PIE): ARPA Piemonte, API bulk paginata, dati orari. Un solo scarico scrive
# PIE (~370 stazioni, meteo e livelli) e PC_PIE (~100 idrometri del Po a monte). Finestra 7 h; nel giro
# delle 03 i 3 giorni interi, per richiudere i buchi.
# GRANDI LAGHI REGOLATI (dati_laghi.py): laghi.net (Maggiore/Como/Iseo/Idro/Garda), serie di 3 giorni
# per sensore, orario a :40. Livello, afflusso, deflusso.
22 * * * *    mkdir -p $BASE/logs && $PYTHON $BASE/dati_pc.py -r PIE >> $BASE/logs/pc-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1
40 * * * *    mkdir -p $BASE/logs && $PYTHON $BASE/dati_laghi.py >> $BASE/logs/laghi-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1

# INVASI ARTIFICIALI (dati_invasi.py): INV_ER Romagna Acque (Ridracoli e traversa del Conca), JSON
# anonimo, aggiornato dalla sorgente ogni 30-60 min.
12,42 * * * * mkdir -p $BASE/logs && $PYTHON $BASE/dati_invasi.py -r all >> $BASE/logs/invasi-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1

# LIGURIA (dati_liguria.py): ARPAL/OMIRL, ora UTC. Istantanee di tutte le classi e serie oraria della
# pioggia; ogni 3 ore le raffiche; nel giro delle 03:14 le serie complete (~3-4 min).
14,44 * * * * mkdir -p $BASE/logs && $PYTHON $BASE/dati_liguria.py >> $BASE/logs/liguria-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1

# --- EXPORT 30 GIORNI ("Confronto 30 giorni"): ogni ora. Scrive solo le serie in site/<REG>/series30/,
# leggendo anche l'archivio compresso (.json.gz); il publish successivo lo pubblica.
48 * * * * mkdir -p $BASE/logs && $PYTHON $BASE/export_web.py -r all --site-dir $BASE/site --days 30 --series-dir series30 --series-only >> $BASE/logs/export30-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1

# --- CLIMATOLOGIA (climatology.py): statistiche e indici climatici dall'archivio completo ->
# site/<REG>/climate.json. Pesante (legge tutto l'archivio, anche i .gz): una volta al giorno, 04:10.
10 4 * * *  mkdir -p $BASE/logs && $PYTHON $BASE/climatology.py -r all --site-dir $BASE/site >> $BASE/logs/climate-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1

# --- PUBLISH ogni 10 min (a :05,:15,... dopo i download). Dati sul branch 'data', pagina su gh-pages
# solo se cambia: si puo' pubblicare spesso senza deploy Pages inutili. Il repository (REMOTE) e' in
# publish.sh; il Pi pubblica via SSH con la sua deploy key.
5,15,25,35,45,55 * * * * mkdir -p $BASE/logs && /bin/bash $BASE/publish.sh >> $BASE/logs/publish-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1

# --- CHECK UPDATES: una volta al giorno, segnala nuove stazioni e variabili non mappate
30 6 * * * mkdir -p $BASE/logs && $PYTHON $BASE/check_updates.py -r all >> $BASE/logs/check-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1

# --- RECOVER (buchi da blackout): @reboot e 4 volte al giorno. Cerca buchi nell'archivio recente e,
# solo se ne trova, scarica lo storico dalle fonti (VEN -date, BZ /timeseries, ER get-time-series,
# LOM Socrata); TN e FVG non hanno storico. Il controllo legge solo file locali: a vuoto costa poco.
# Lo 'sleep 120' a reboot da' tempo alla rete di salire.
@reboot sleep 120 && mkdir -p $BASE/logs && $PYTHON $BASE/recover.py -r all >> $BASE/logs/recover-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1
40 0,6,12,18 * * * mkdir -p $BASE/logs && $PYTHON $BASE/recover.py -r all >> $BASE/logs/recover-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1

# --- PULIZIA notturna (03:15): log piu' vecchi di 14 giorni e grezzi orfani piu' vecchi di 2 giorni
# (rimasti da un parse fallito). Non tocca data-parsed/ ne' site/.
15 3 * * * find $BASE/logs $BASE/*/logs -name '*.log*' -mtime +14 -delete 2>/dev/null ; find $BASE/*/data-downloaded -type f -mtime +2 -delete 2>/dev/null ; find $BASE/*/data-parsed $BASE/site -name '*.tmp' -mtime +1 -delete 2>/dev/null

# --- COMPRESSIONE archivio (03:30): gzip dei file data-parsed piu' vecchi di 8 giorni. L'archivio
# resta completo; i file recenti, usati da export_web e da recover, restano non compressi. gzip salta
# i .gz. Per rileggere un file: gunzip <file>.json.gz (o zcat).
30 3 * * * find $BASE/*/data-parsed -name '*-*.json' -mtime +8 -exec gzip -q {} \; 2>/dev/null
# Comprime anche i log piu' vecchi di 1 giorno (testo, ~15x): 14 giorni di storico occupano poco.
35 3 * * * find $BASE/logs $BASE/*/logs -name '*.log' -mtime +1 -exec gzip -q {} \; 2>/dev/null

# --- BACKUP sulla chiavetta USB (04:40: dopo compressione e climatologia, prima del riavvio 05:50).
# backup_usb.sh salva ogni notte una versione completa di METEO (senza site/, log, grezzi) e di ARPAV
# in /mnt/usb/backup/AAAA-MM-GG/ e tiene le ultime 30; i file uguali sono hard link. Chiavetta non
# montata: non scrive nulla. Mai due insieme (flock nello script). Controllo: ~/METEO/check_size.sh.
40 4 * * * mkdir -p $BASE/logs && /bin/bash $BASE/backup_usb.sh >> $BASE/logs/backup-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1

# --- STORICO LUNGO (07:20): history.py aggrega pioggia e temperatura giornaliere dell'archivio e
# publish_history.sh li pubblica sul branch 'history' (scheda "Storico"). Rielabora i giorni nuovi e
# gli ultimi 8 (cache in history_cache/). Alle 07:20 il giorno prima e' arrivato anche dalle fonti
# lente (Lombardia pubblica con 4-5 ore di ritardo). flock: mai due giri insieme.
20 7 * * * mkdir -p $BASE/logs && flock -n $BASE/.history.lock /bin/bash $BASE/publish_history.sh >> $BASE/logs/history-`date +\%Y\%m\%d\%H\%M\%S`.log 2>&1

# --- RADAR DPC (dati_radar.py): radar SRI della Protezione Civile, ultime 6 ore in PNG per la mappa,
# sul branch 'radar' (letto da site/radar.js). Ogni 5 min a :03/:08/...: il DPC pubblica ogni
# fotogramma ~7-8 min dopo l'istante. flock -n: mai due insieme. Un log al giorno (288 giri); la
# pulizia notturna dei log vale anche qui.
3-58/5 * * * * mkdir -p $BASE/logs && flock -n $BASE/.radar.lock $PYTHON $BASE/dati_radar.py >> $BASE/logs/radar-`date +\%Y\%m\%d`.log 2>&1
